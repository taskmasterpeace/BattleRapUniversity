```tsx
// components/media/BloggerAvatar.tsx

interface BloggerAvatarProps {
  blogger: BloggerName;
  size: 'sm' | 'md' | 'lg' | 'xl';
  showBadge?: boolean;  // Show specialty badge
  onClick?: () => void;
}

// Size mappings
const SIZES = {
  sm: 'w-8 h-8',      // 32px - inline mentions
  md: 'w-12 h-12',    // 48px - article bylines
  lg: 'w-16 h-16',    // 64px - blogger cards
  xl: 'w-24 h-24',    // 96px - blogger profile page
};

// Render:
<div className={`${SIZES[size]} rounded-full overflow-hidden border-2`}
     style={{ borderColor: BLOGGER_COLORS[blogger] }}>
  <img src={`/avatars/bloggers/${blogger.slug}.png`}
       alt={blogger.name}
       className="w-full h-full object-cover" />
  {showBadge && (
    <span className="absolute -bottom-1 -right-1 text-xs">
      {BLOGGER_ICONS[blogger]}
    </span>
  )}
</div>
```

## 2.4 BLOGGER SELECTOR COMPONENT

Dropdown or pill selector to filter articles by blogger.

```tsx
// components/media/BloggerSelector.tsx

interface BloggerSelectorProps {
  selected: BloggerName | 'all';
  onChange: (blogger: BloggerName | 'all') => void;
  variant: 'pills' | 'dropdown';
}

// PILLS variant (horizontal scrollable):
┌────────────────────────────────────────────────────────────────────┐
│ [All] [👁️ Battle Eyez] [🔥 MP] [📊 AI] [🎤 SRR] [👑 Herald] ...  │
└────────────────────────────────────────────────────────────────────┘

// Render (pills):
<div className="flex gap-2 overflow-x-auto pb-2 hide-scrollbar">
  <button
    onClick={() => onChange('all')}
    className={`px-4 py-2 rounded-full font-bold uppercase text-sm whitespace-nowrap
                ${selected === 'all'
                  ? 'bg-orange-500 text-black'
                  : 'bg-zinc-800 text-zinc-300 hover:bg-zinc-700'}`}
  >
    All Bloggers
  </button>

  {BLOGGERS.map(blogger => (
    <button
      key={blogger.slug}
      onClick={() => onChange(blogger.name)}
      className={`px-4 py-2 rounded-full font-bold uppercase text-sm
                  whitespace-nowrap flex items-center gap-2
                  ${selected === blogger.name
                    ? 'text-black'
                    : 'bg-zinc-800 text-zinc-300 hover:bg-zinc-700'}`}
      style={selected === blogger.name ? { backgroundColor: blogger.color } : {}}
    >
      <span>{blogger.icon}</span>
      <span>{blogger.name}</span>
    </button>
  ))}
</div>

// DROPDOWN variant:
<select
  value={selected}
  onChange={(e) => onChange(e.target.value)}
  className="bg-zinc-900 border border-zinc-700 text-white px-4 py-2"
>
  <option value="all">All Bloggers</option>
  {BLOGGERS.map(blogger => (
    <option key={blogger.slug} value={blogger.name}>
      {blogger.icon} {blogger.name}
    </option>
  ))}
</select>
```
